<script>
const $=s=>document.querySelector(s);
const PG=[["home","Home"],["ask","Ask Humanity"],["about","About Oracle"]];
const V={
home(){return`<h1>The World’s Knowledge. Humanity’s Perspectives.</h1><p>Explore the ideas, beliefs, traditions and wisdom that have shaped humanity.</p><div class="card" onclick="location.hash='ask'"><h3>ASK HUMANITY</h3><p>An open conversational experience.</p></div>`},
ask(){return`<h1>Ask Humanity</h1><p>Ask about beliefs, philosophy, spirituality or existence.</p><div class="card"><textarea id="qi" placeholder="For example: What does the text say about love?"></textarea><p><button class="btn" onclick="doAsk()">Ask</button></p><div id="out"></div></div>`},
about(){return`<h1>About Oracle</h1><p>The objective is to help humanity become better at being different together.</p>`}
};
async function doAsk(){
    const q=$("#qi").value.trim(); if(!q) return;
    const out=$("#out"); out.innerHTML="<p>Searching...</p>";
    try{
        const res=await fetch("/api/ask",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({question:q})});
        if(!res.ok) throw new Error("Backend offline");
        const data=await res.json();
        let html=`<div class="box s"><h3>Universal thread</h3><p>${data.universal_thread||"No shared thread found."}</p></div>`;
        if(data.aligned_verses&&data.aligned_verses.length>0){
            html+=`<div class="box"><h3>Aligned verses</h3>`;
            data.aligned_verses.forEach(v=>{html+=`<p><b>${v.faith} (${v.citation}):</b> “${v.text}”</p>`});
            html+=`</div>`;
        }
        if(data.distinct_rationales&&data.distinct_rationales.length>0){
            html+=`<div class="box s"><h3>Distinct rationales</h3>`;
            data.distinct_rationales.forEach(r=>{html+=`<p><b>${r.faith}:</b> ${r.theological_logic}</p>`});
            html+=`</div>`;
        }
        out.innerHTML=html;
    }catch(err){
        out.innerHTML=`<p style="color:var(--bd)">Error: ${err.message}</p>`;
    }
}
function render(){const id=(location.hash||"#home").slice(1),p=V[id]?id:"home";$("#nv").innerHTML=PG.map(x=>`<a href="#${x[0]}" class="${x[0]===p?"on":""}">${x[1]}</a>`).join("");$("#m").innerHTML=V[p]()}
addEventListener("hashchange",render);render();
</script>
